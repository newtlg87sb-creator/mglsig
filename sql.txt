-- ============================================================================
-- 1. USER & PROFILE MANAGEMENT (Хэрэглэгчийн удирдлага)
-- ============================================================================

-- 'User control' хүснэгт: Хэрэглэгчийн профайл, гишүүнчлэл
CREATE TABLE IF NOT EXISTS public."User control" (
  id uuid PRIMARY KEY REFERENCES auth.users ON DELETE CASCADE,
  username text,
  email text,
  membership_type text DEFAULT 'free',
  membership_expires_at timestamptz,
  updated_at timestamptz DEFAULT now()
);

-- Автоматаар профайл үүсгэх функц (Auth Trigger)
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO public."User control" (id, username, email, membership_type)
  VALUES (
    new.id, 
    COALESCE(new.raw_user_meta_data->>'display_name', SPLIT_PART(new.email, '@', 1)), 
    new.email, 
    'free'
  );
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Триггер идэвхжүүлэх
DROP TRIGGER IF EXISTS on_auth_user_created ON auth.users;
CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE PROCEDURE public.handle_new_user();

ALTER TABLE public."User control" ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Allow individual read" ON public."User control";
CREATE POLICY "Allow individual read" ON public."User control" FOR SELECT USING (auth.uid() = id);

-- Kucoin Market Data
CREATE TABLE IF NOT EXISTS market_data (
  symbol text PRIMARY KEY,
  full_symbol text,
  bid float8,
  ask float8,
  spread float8,
  min_usdt float8,
  volume float8,
  real_change float8,
  h1_change float8,
  change_24h float8,
  is_st boolean DEFAULT false,
  min_order_cost float8,
  updated_at timestamptz DEFAULT now()
);

-- Binance Market Data
CREATE TABLE IF NOT EXISTS binance_market_data (
  symbol text PRIMARY KEY,
  full_symbol text,
  bid float8,
  ask float8,
  spread float8,
  min_usdt float8,
  volume float8,
  real_change float8,
  h1_change float8,
  change_24h float8,

  min_order_cost float8,
  updated_at timestamptz DEFAULT now()
);

ALTER TABLE market_data ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Enable read access for all users" ON market_data;

ALTER TABLE binance_market_data ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Enable read access for all users on binance" ON binance_market_data;
CREATE POLICY "Enable read access for all users on binance" ON binance_market_data FOR SELECT USING (true);

-- ЧУХАЛ: Realtime-ийг унтраах (Limit хэтрэхээс сэргийлнэ)
DO $$
BEGIN
  -- market_data-г хасах (байвал)
  IF EXISTS (SELECT 1 FROM pg_publication_tables WHERE pubname = 'supabase_realtime' AND tablename = 'market_data') THEN
    ALTER PUBLICATION supabase_realtime DROP TABLE market_data;
  END IF;

  -- binance_market_data-г хасах (байвал)
  IF EXISTS (SELECT 1 FROM pg_publication_tables WHERE pubname = 'supabase_realtime' AND tablename = 'binance_market_data') THEN
    ALTER PUBLICATION supabase_realtime DROP TABLE binance_market_data;
  END IF;
END $$;
